
      alert(
        "Erro ao cadastrar morador."
      );
    } finally {
      setSalvandoMorador(
        false
      );
    }
  }

    prioridade: number;

    podeAbrirPortao: boolean;

      "Morador nao informado."
    );
  }

  const nome =
    dados.nome.trim();

  const telefone =
    dados.telefone.trim();

  if (!nome) {
    throw new Error(
      "Informe o nome do morador."
    );
  }

  if (!telefone) {
    throw new Error(
      "Informe o telefone do morador."
    );
  }

  await update(
    ref(
      db,
      `qrCentral/moradores/${moradorId}`
    ),
    {
      nome:
        formatarNome(nome),

      telefone,

      email:
        dados.email?.trim() || "",

      prioridade:
        Number(dados.prioridade),

      ordemAtendimento:
        Number(dados.prioridade),

      recebeChamadas:
        dados.recebeChamadas,

      podeAbrirPortao:
        dados.podeAbrirPortao,

      status:
        dados.status,

      atualizadoEm:
        new Date().toISOString(),

      ultimoStatusEm:
        Date.now(),
    }
  );
