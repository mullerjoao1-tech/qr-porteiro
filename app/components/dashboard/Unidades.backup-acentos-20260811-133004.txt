"use client";

import { useMemo, useState } from "react";
import ResponsaveisUnidade from "./ResponsaveisUnidade";

type LocalCadastrado = {
  id: string;
  nome: string;
  tipo: string;
};

type UnidadeCadastrada = {
  id: string;
  codigo: string;
  localId: string;
  localNome: string;
  tipo: string;
  bloco: string;
  nome: string;
  modoChamado?: string;
};

type AbaUnidade =
  | "geral"
  | "responsaveis"
  | "historico"
  | "visitantes"
  | "entregas"
  | "acessos"
  | "documentos"
  | "configuracoes";

type Props = {
  locais: LocalCadastrado[];
  unidades: UnidadeCadastrada[];
  localSelecionadoId: string;
  setLocalSelecionadoId: (valor: string) => void;
  blocoUnidade: string;
  setBlocoUnidade: (valor: string) => void;
  nomeUnidade: string;
  setNomeUnidade: (valor: string) => void;
  tipoUnidade: string;
  setTipoUnidade: (valor: string) => void;
  modoChamadoUnidade: string;
  setModoChamadoUnidade: (valor: string) => void;
  modoCondominio: boolean;
  cadastrarUnidade: () => void;
  salvandoUnidade: boolean;
};

export default function Unidades({
  locais,
  unidades,
  localSelecionadoId,
  setLocalSelecionadoId,
  blocoUnidade,
  setBlocoUnidade,
  nomeUnidade,
  setNomeUnidade,
  tipoUnidade,
  setTipoUnidade,
  modoChamadoUnidade,
  setModoChamadoUnidade,
  modoCondominio,
  cadastrarUnidade,
  salvandoUnidade,
}: Props) {
  const [busca, setBusca] = useState("");
  const [filtroLocal, setFiltroLocal] = useState("todos");
  const [filtroTipo, setFiltroTipo] = useState("todos");
  const [modalCadastroAberto, setModalCadastroAberto] = useState(false);

const contextoLocalTravado =
  modoCondominio &&
  Boolean(localSelecionadoId);
  const [unidadeSelecionada, setUnidadeSelecionada] =
    useState<UnidadeCadastrada | null>(null);
  const [abaAtiva, setAbaAtiva] = useState<AbaUnidade>("geral");

  const unidadesFiltradas = useMemo(() => {
    const termo = busca.trim().toLowerCase();

    return unidades
      .filter((unidade) => {
        const passaBusca =
          !termo ||
          [
            unidade.codigo,
            unidade.localNome,
            unidade.bloco,
            unidade.nome,
            unidade.tipo,
            unidade.modoChamado,
          ]
            .filter(Boolean)
            .join(" ")
            .toLowerCase()
            .includes(termo);

        const passaLocal =
          filtroLocal === "todos" || unidade.localId === filtroLocal;

        const passaTipo =
          filtroTipo === "todos" || unidade.tipo === filtroTipo;

        return passaBusca && passaLocal && passaTipo;
      })
      .sort((a, b) => {
        const local = a.localNome.localeCompare(b.localNome, "pt-BR");

        if (local !== 0) return local;

        const bloco = String(a.bloco || "").localeCompare(
          String(b.bloco || ""),
          "pt-BR",
          { numeric: true }
        );

        if (bloco !== 0) return bloco;

        return String(a.nome).localeCompare(String(b.nome), "pt-BR", {
          numeric: true,
        });
      });
  }, [busca, filtroLocal, filtroTipo, unidades]);

  const totalApartamentos = unidades.filter(
    (unidade) => unidade.tipo === "apartamento"
  ).length;

  const totalFamilia = unidades.filter(
    (unidade) => unidade.modoChamado !== "prioridade"
  ).length;

  const totalPrioridade = unidades.filter(
    (unidade) => unidade.modoChamado === "prioridade"
  ).length;

  const tiposUnidades = Array.from(
    new Set(unidades.map((unidade) => unidade.tipo).filter(Boolean))
  ).sort((a, b) => a.localeCompare(b, "pt-BR"));

  function textoModoChamado(modo?: string) {
    if (modo === "prioridade") return "Modo Prioridade";
    return "Modo Família";
  }

  function textoTipoUnidade(tipo: string) {
    const tipos: Record<string, string> = {
      apartamento: "Apartamento",
      sala: "Sala",
      loja: "Loja",
      livre: "Livre",
      casa: "Casa",
      quarto: "Quarto",
      chale: "Chalé",
    };

    return tipos[tipo] || tipo;
  }

  function nomeCompletoUnidade(unidade: UnidadeCadastrada) {
    return unidade.bloco
      ? `${unidade.bloco} / ${unidade.nome}`
      : unidade.nome;
  }

  function abrirUnidade(unidade: UnidadeCadastrada) {
    setUnidadeSelecionada(unidade);
    setAbaAtiva("geral");
  }

  function fecharUnidade() {
    setUnidadeSelecionada(null);
    setAbaAtiva("geral");
  }

  function fecharCadastro() {
    if (salvandoUnidade) return;
    setModalCadastroAberto(false);
  }

  const abas: Array<{
    id: AbaUnidade;
    nome: string;
    icone: string;
  }> = [
    { id: "geral", nome: "Geral", icone: "🏠" },
    { id: "responsaveis", nome: "Responsáveis", icone: "👥" },
    { id: "historico", nome: "Histórico", icone: "🕘" },
    { id: "visitantes", nome: "Visitantes", icone: "🚶" },
    { id: "entregas", nome: "Entregas", icone: "📦" },
    { id: "acessos", nome: "Acessos", icone: "🚪" },
    { id: "documentos", nome: "Documentos", icone: "📄" },
    { id: "configuracoes", nome: "Configurações", icone: "⚙️" },
  ];

  return (
    <div className="space-y-5">
      {/* Cabeçalho */}

      <section className="rounded-3xl bg-gradient-to-r from-blue-700 to-cyan-600 p-5 text-white md:p-7">
        <div className="flex flex-col gap-5 md:flex-row md:items-center md:justify-between">
          <div>
            <p className="text-sm font-bold text-blue-100">
              🏠 MÓDULO DE UNIDADES
            </p>

            <h2 className="mt-1 text-3xl font-black md:text-4xl">
              Unidades
            </h2>

            <p className="mt-2 text-sm text-blue-100 md:text-base">
              Gerencie apartamentos, casas, salas, lojas, quartos, chalés e
              demais unidades vinculadas aos locais.
            </p>
          </div>

          <button
            type="button"
            onClick={() => setModalCadastroAberto(true)}
            className="rounded-2xl bg-white px-5 py-3 font-black text-blue-700 shadow-lg transition-all hover:bg-blue-50 active:scale-95"
          >
            ＋ Nova unidade
          </button>
        </div>
      </section>

      {/* Indicadores */}

      <section className="grid grid-cols-2 gap-3 lg:grid-cols-4">
        <div className="rounded-2xl border border-slate-700 bg-slate-900 p-4">
          <p className="text-xs font-bold text-slate-400">
            TOTAL CADASTRADO
          </p>
          <p className="mt-2 text-3xl font-black text-white">
            {unidades.length}
          </p>
        </div>

        <div className="rounded-2xl border border-blue-800 bg-blue-950/25 p-4">
          <p className="text-xs font-bold text-blue-300">
            APARTAMENTOS
          </p>
          <p className="mt-2 text-3xl font-black text-white">
            {totalApartamentos}
          </p>
        </div>

        <div className="rounded-2xl border border-green-800 bg-green-950/25 p-4">
          <p className="text-xs font-bold text-green-300">
            MODO FAMÍLIA
          </p>
          <p className="mt-2 text-3xl font-black text-white">
            {totalFamilia}
          </p>
        </div>

        <div className="rounded-2xl border border-orange-800 bg-orange-950/25 p-4">
          <p className="text-xs font-bold text-orange-300">
            MODO PRIORIDADE
          </p>
          <p className="mt-2 text-3xl font-black text-white">
            {totalPrioridade}
          </p>
        </div>
      </section>

      {/* Pesquisa e filtros */}

      <section className="rounded-2xl border border-slate-800 bg-slate-900 p-4 md:p-5">
        <div className="grid gap-3 lg:grid-cols-[1fr_260px_210px]">
          <div className="relative">
            <span className="pointer-events-none absolute left-4 top-1/2 -translate-y-1/2 text-lg">
              🔍
            </span>

            <input
              value={busca}
              onChange={(event) => setBusca(event.target.value)}
              placeholder="Pesquisar por unidade, bloco, local ou código..."
              className="w-full rounded-xl border border-slate-700 bg-slate-800 py-3 pl-12 pr-4 text-white outline-none transition-all placeholder:text-slate-500 focus:border-blue-500"
            />
          </div>

          <select
            value={filtroLocal}
            onChange={(event) => setFiltroLocal(event.target.value)}
            className="w-full rounded-xl border border-slate-700 bg-slate-800 p-3 text-white outline-none focus:border-blue-500"
          >
            <option value="todos">🌐 Todos os locais</option>

            {locais.map((local) => (
              <option key={local.id} value={local.id}>
                🏢 {local.nome}
              </option>
            ))}
          </select>

          <select
            value={filtroTipo}
            onChange={(event) => setFiltroTipo(event.target.value)}
            className="w-full rounded-xl border border-slate-700 bg-slate-800 p-3 text-white outline-none focus:border-blue-500"
          >
            <option value="todos">Todos os tipos</option>

            {tiposUnidades.map((tipo) => (
              <option key={tipo} value={tipo}>
                {textoTipoUnidade(tipo)}
              </option>
            ))}
          </select>
        </div>

        <div className="mt-4 flex flex-wrap items-center justify-between gap-3">
          <p className="text-sm text-slate-400">
            {unidadesFiltradas.length} unidade
            {unidadesFiltradas.length === 1 ? "" : "s"} encontrada
            {unidadesFiltradas.length === 1 ? "" : "s"}.
          </p>

          {(busca || filtroLocal !== "todos" || filtroTipo !== "todos") && (
            <button
              type="button"
              onClick={() => {
                setBusca("");
                setFiltroLocal("todos");
                setFiltroTipo("todos");
              }}
              className="rounded-xl bg-slate-800 px-4 py-2 text-sm font-black text-slate-300 transition-all hover:bg-slate-700 active:scale-95"
            >
              Limpar filtros
            </button>
          )}
        </div>
      </section>

      {/* Lista */}

      <section className="rounded-2xl border border-slate-800 bg-slate-900 p-4 md:p-5">
        <div>
          <p className="text-xs font-bold text-blue-300">
            LISTA DE UNIDADES
          </p>

          <h3 className="mt-1 text-2xl font-black text-white">
            Selecione uma unidade
          </h3>
        </div>

        {unidadesFiltradas.length === 0 ? (
          <div className="mt-5 rounded-2xl border border-slate-700 bg-slate-800 p-8 text-center">
            <div className="text-4xl">🏠</div>
            <p className="mt-3 font-black text-slate-300">
              Nenhuma unidade encontrada
            </p>
            <p className="mt-2 text-sm text-slate-500">
              Ajuste os filtros ou cadastre uma nova unidade.
            </p>
          </div>
        ) : (
          <div className="mt-5 grid gap-3 lg:grid-cols-2">
            {unidadesFiltradas.map((unidade) => (
              <button
                key={unidade.id}
                type="button"
                onClick={() => abrirUnidade(unidade)}
                className="rounded-2xl border border-slate-700 bg-slate-800 p-4 text-left transition-all hover:border-blue-500 hover:bg-slate-700 active:scale-[0.98]"
              >
                <div className="flex items-start justify-between gap-4">
                  <div className="flex min-w-0 items-start gap-3">
                    <div className="flex h-12 w-12 shrink-0 items-center justify-center rounded-2xl bg-blue-950 text-2xl">
                      🏠
                    </div>

                    <div className="min-w-0">
                      <p className="text-xs font-black text-blue-300">
                        {unidade.codigo}
                      </p>

                      <h4 className="mt-1 truncate text-lg font-black text-white">
                        {nomeCompletoUnidade(unidade)}
                      </h4>

                      <p className="mt-1 text-sm text-slate-300">
                        🏢 {unidade.localNome}
                      </p>

                      <p className="mt-1 text-sm text-slate-400">
                        {textoTipoUnidade(unidade.tipo)}
                      </p>
                    </div>
                  </div>

                  <div className="shrink-0 text-right">
                    <span
                      className={`inline-flex rounded-full px-3 py-1 text-[10px] font-black ${
                        unidade.modoChamado === "prioridade"
                          ? "bg-orange-950 text-orange-300"
                          : "bg-green-950 text-green-300"
                      }`}
                    >
                      {unidade.modoChamado === "prioridade"
                        ? "🟠 Prioridade"
                        : "🟢 Família"}
                    </span>

                    <p className="mt-3 text-xs font-bold text-slate-300">
                      👥 Moradores vinculados
                    </p>

                    <p className="mt-1 text-xs font-bold text-slate-400">
                      🕘 Histórico da unidade
                    </p>

                    <p className="mt-3 text-xs font-black text-blue-300">
                      Abrir perfil →
                    </p>
                  </div>
                </div>
              </button>
            ))}
          </div>
        )}
      </section>

      {/* Cadastro */}

      {modalCadastroAberto && (
        <div className="fixed inset-0 z-[130] flex items-center justify-center bg-black/75 p-3 md:p-6">
          <div className="max-h-[92vh] w-full max-w-2xl overflow-y-auto rounded-3xl border border-slate-700 bg-slate-900 p-4 shadow-2xl md:p-6">
            <div className="flex items-start justify-between gap-4">
              <div>
                <p className="text-sm font-bold text-blue-300">
                  ＋ NOVA UNIDADE
                </p>

                <h3 className="mt-1 text-2xl font-black text-white">
                  Cadastro de unidade
                </h3>

                <p className="mt-2 text-sm text-slate-400">
                  Cadastre a unidade deste condom?nio e defina seu modo de chamada.
                </p>
              </div>

              <button
                type="button"
                onClick={fecharCadastro}
                disabled={salvandoUnidade}
                className="flex h-11 w-11 shrink-0 items-center justify-center rounded-xl bg-slate-800 text-xl font-black transition-all hover:bg-slate-700 active:scale-95 disabled:opacity-50"
              >
                ✕
              </button>
            </div>

            <div className="mt-5 space-y-4">
              {!contextoLocalTravado && (
              <select
                value={localSelecionadoId}
                onChange={(event) =>
                  setLocalSelecionadoId(event.target.value)
                }
                className="w-full rounded-xl border border-slate-700 bg-slate-800 p-3 text-white outline-none focus:border-blue-500"
              >
                <option value="">Selecione o local</option>

                {locais.map((local) => (
                  <option key={local.id} value={local.id}>
                    {local.nome}
                  </option>
                ))}
              </select>
              )}

              {modoCondominio && (
                <input
                  value={blocoUnidade}
                  onChange={(event) =>
                    setBlocoUnidade(event.target.value)
                  }
                  placeholder="Bloco ou torre. Ex: Bloco A"
                  className="w-full rounded-xl border border-slate-700 bg-slate-800 p-3 text-white outline-none placeholder:text-slate-500 focus:border-blue-500"
                />
              )}

              <input
                value={nomeUnidade}
                onChange={(event) =>
                  setNomeUnidade(event.target.value)
                }
                placeholder={
                  modoCondominio
                    ? "Número da unidade. Ex: 101"
                    : "Nome livre"
                }
                className="w-full rounded-xl border border-slate-700 bg-slate-800 p-3 text-white outline-none placeholder:text-slate-500 focus:border-blue-500"
              />

              {!contextoLocalTravado && (
              <select
                value={tipoUnidade}
                onChange={(event) =>
                  setTipoUnidade(event.target.value)
                }
                className="w-full rounded-xl border border-slate-700 bg-slate-800 p-3 text-white outline-none focus:border-blue-500"
              >
                {modoCondominio ? (
                  <>
                    <option value="apartamento">Apartamento</option>
                    <option value="sala">Sala</option>
                    <option value="loja">Loja</option>
                  </>
                ) : (
                  <>
                    <option value="livre">Livre</option>
                    <option value="casa">Casa</option>
                    <option value="quarto">Quarto</option>
                    <option value="chale">Chalé</option>
                  </>
                )}
              </select>
              )}

              <select
                value={modoChamadoUnidade}
                onChange={(event) =>
                  setModoChamadoUnidade(event.target.value)
                }
                className="w-full rounded-xl border border-slate-700 bg-slate-800 p-3 text-white outline-none focus:border-blue-500"
              >
                <option value="familia">
                  Modo Família — todos juntos
                </option>
                <option value="prioridade">
                  Modo Prioridade — em fila
                </option>
              </select>

              <div className="rounded-xl border border-blue-900 bg-blue-950/25 p-4 text-sm text-blue-200">
                O modo Família chama todos os moradores juntos. O modo
                Prioridade respeita a ordem definida no cadastro dos moradores.
              </div>

              <div className="grid grid-cols-2 gap-3">
                <button
                  type="button"
                  onClick={fecharCadastro}
                  disabled={salvandoUnidade}
                  className="rounded-xl bg-slate-700 py-3 font-black text-white transition-all hover:bg-slate-600 active:scale-95 disabled:opacity-50"
                >
                  Cancelar
                </button>

                <button
                  type="button"
                  onClick={cadastrarUnidade}
                  disabled={salvandoUnidade}
                  className="rounded-xl bg-blue-600 py-3 font-black text-white transition-all hover:bg-blue-500 active:scale-95 disabled:bg-slate-600"
                >
                  {salvandoUnidade
                    ? "Salvando..."
                    : "Cadastrar unidade"}
                </button>
              </div>
            </div>
          </div>
        </div>
      )}

      {/* Perfil */}

      {unidadeSelecionada && (
        <div className="fixed inset-0 z-[125] flex items-center justify-center bg-black/75 p-3 md:p-6">
          <div className="max-h-[94vh] w-full max-w-5xl overflow-y-auto rounded-3xl border border-slate-700 bg-slate-900 p-4 shadow-2xl md:p-6">
            <div className="flex items-start justify-between gap-4">
              <div className="flex min-w-0 items-start gap-4">
                <div className="flex h-14 w-14 shrink-0 items-center justify-center rounded-2xl bg-blue-950 text-3xl">
                  🏠
                </div>

                <div className="min-w-0">
                  <p className="text-xs font-black text-blue-300">
                    {unidadeSelecionada.codigo}
                  </p>

                  <h3 className="mt-1 truncate text-2xl font-black text-white md:text-3xl">
                    {nomeCompletoUnidade(unidadeSelecionada)}
                  </h3>

                  <p className="mt-1 text-sm text-slate-400">
                    🏢 {unidadeSelecionada.localNome} •{" "}
                    {textoTipoUnidade(unidadeSelecionada.tipo)}
                  </p>
                </div>
              </div>

              <button
                type="button"
                onClick={fecharUnidade}
                className="flex h-11 w-11 shrink-0 items-center justify-center rounded-xl bg-slate-800 text-xl font-black transition-all hover:bg-slate-700 active:scale-95"
              >
                ✕
              </button>
            </div>

            <div className="mt-5 grid grid-cols-2 gap-2 md:hidden">
              {abas.map((aba) => (
                <button
                  key={aba.id}
                  type="button"
                  onClick={() => setAbaAtiva(aba.id)}
                  className={`min-h-[48px] rounded-xl px-3 py-3 text-left text-xs font-black transition-all active:scale-95 ${
                    abaAtiva === aba.id
                      ? "bg-blue-600 text-white"
                      : "bg-slate-800 text-slate-300"
                  }`}
                >
                  <span className="mr-2">{aba.icone}</span>
                  {aba.nome}
                </button>
              ))}
            </div>

            <div className="mt-5 hidden gap-2 overflow-x-auto pb-2 md:flex">
              {abas.map((aba) => (
                <button
                  key={aba.id}
                  type="button"
                  onClick={() => setAbaAtiva(aba.id)}
                  className={`shrink-0 rounded-xl px-4 py-3 text-sm font-black transition-all active:scale-95 ${
                    abaAtiva === aba.id
                      ? "bg-blue-600 text-white"
                      : "bg-slate-800 text-slate-300 hover:bg-slate-700"
                  }`}
                >
                  {aba.icone} {aba.nome}
                </button>
              ))}
            </div>

            {abaAtiva === "geral" && (
              <div className="mt-5 space-y-4">
                <div className="grid gap-4 lg:grid-cols-3">
                  <div className="rounded-2xl border border-slate-700 bg-slate-800 p-4 lg:col-span-2">
                    <p className="text-xs font-bold text-slate-400">
                      DADOS PRINCIPAIS
                    </p>

                    <div className="mt-4 grid gap-4 sm:grid-cols-2">
                      <div>
                        <p className="text-xs text-slate-500">Local</p>
                        <p className="mt-1 font-black text-white">
                          {unidadeSelecionada.localNome}
                        </p>
                      </div>

                      <div>
                        <p className="text-xs text-slate-500">
                          Unidade
                        </p>
                        <p className="mt-1 font-black text-white">
                          {nomeCompletoUnidade(unidadeSelecionada)}
                        </p>
                      </div>

                      <div>
                        <p className="text-xs text-slate-500">Tipo</p>
                        <p className="mt-1 font-black text-white">
                          {textoTipoUnidade(unidadeSelecionada.tipo)}
                        </p>
                      </div>

                      <div>
                        <p className="text-xs text-slate-500">
                          Modo de chamada
                        </p>
                        <p
                          className={`mt-1 font-black ${
                            unidadeSelecionada.modoChamado === "prioridade"
                              ? "text-orange-300"
                              : "text-green-300"
                          }`}
                        >
                          {textoModoChamado(
                            unidadeSelecionada.modoChamado
                          )}
                        </p>
                      </div>
                    </div>
                  </div>

                  <div className="rounded-2xl border border-cyan-800 bg-cyan-950/25 p-4">
                    <p className="text-xs font-bold text-cyan-300">
                      🤖 RESUMO
                    </p>

                    <p className="mt-3 text-sm leading-relaxed text-slate-200">
                      Unidade vinculada ao local{" "}
                      {unidadeSelecionada.localNome}, configurada em{" "}
                      {textoModoChamado(
                        unidadeSelecionada.modoChamado
                      ).toLowerCase()}.
                    </p>

                    <div className="mt-4 rounded-xl bg-slate-900 p-3">
                      <p className="text-xs text-slate-500">
                        Situação atual
                      </p>
                      <p className="mt-1 font-black text-white">
                        Estrutura pronta para gestão
                      </p>
                    </div>
                  </div>
                </div>

                <div className="grid gap-4 lg:grid-cols-[1.15fr_0.85fr]">
                  <div className="rounded-2xl border border-slate-700 bg-slate-800 p-4">
                    <p className="text-xs font-bold text-slate-400">
                      RESUMO DA UNIDADE
