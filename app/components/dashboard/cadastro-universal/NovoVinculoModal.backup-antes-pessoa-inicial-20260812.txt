"use client";

import {
  useEffect,
  useMemo,
  useState,
} from "react";

import {
  onValue,
  ref,
  update,
} from "firebase/database";

import {
  db,
} from "../../../services/firebase";

type Pessoa = {
  id: string;
  nome?: string;
  email?: string;

  locais?: Record<
    string,
    unknown
  >;

  condominios?: Record<
    string,
    unknown
  >;
};

type Local = {
  id: string;
  nome?: string;
  slug?: string;
  tipoLocal?: string;
};

type Unidade = {
  id: string;
  localId?: string;
  condominioId?: string;
  bloco?: string;
  nome?: string;
  codigo?: string;
};

type Props = {
  aberto: boolean;
  onClose: () => void;
};

const PERFIS = [
  "morador",
  "sindico",
  "administradora",
  "gestor_local",
  "porteiro",
  "central",
  "funcionario",
  "financeiro",
  "prestador",
  "conselheiro",
  "administrador_master",
  "proprietario",
  "inquilino",
  "responsavel",
  "gerente",
  "outro",
];

function nomePerfil(
  valor: string
) {
  const nomes: Record<
    string,
    string
  > = {
    morador: "Morador",
    sindico: "Síndico",
    administradora: "Administradora",
    gestor_local: "Gestor local",
    porteiro: "Porteiro",
    central: "Central",
    funcionario: "Funcionário",
    financeiro: "Financeiro",
    prestador: "Prestador",
    conselheiro: "Conselheiro",
    administrador_master:
      "Administrador master",
    proprietario: "Proprietário",
    inquilino: "Inquilino",
    responsavel: "Responsável",
    gerente: "Gerente",
    outro: "Outro",
  };

  return nomes[valor] || valor;
}

function nomeUnidade(
  unidade: Unidade
) {
  const nome =
    unidade.nome ||
    unidade.codigo ||
    unidade.id;

  return unidade.bloco
    ? `${unidade.bloco} • ${nome}`
    : nome;
}

type VinculoExistente = {
  localId?: string;
  localSlug?: string;
  condominioId?: string;
  condominioSlug?: string;
  slug?: string;

  perfilPrincipal?: string;

  perfis?: Record<
    string,
    boolean
  >;

  unidades?: Record<
    string,
    boolean
  >;

  permissoes?: Record<
    string,
    boolean
  >;

  ativo?: boolean;

  criadoEm?: number;
  atualizadoEm?: number;

  [chave: string]: unknown;
};
export default function NovoVinculoModal({
  aberto,
  onClose,
}: Props) {
  const [
    pessoas,
    setPessoas,
  ] = useState<Pessoa[]>([]);

  const [
    locais,
    setLocais,
  ] = useState<Local[]>([]);

  const [
    unidades,
    setUnidades,
  ] = useState<Unidade[]>([]);

  const [
    pessoaId,
    setPessoaId,
  ] = useState("");

  const [
    localId,
    setLocalId,
  ] = useState("");

  const [
    perfis,
    setPerfis,
  ] = useState<string[]>([]);

  const [
    perfilPadrao,
    setPerfilPadrao,
  ] = useState("");

  const [
    unidadesSelecionadas,
    setUnidadesSelecionadas,
  ] = useState<string[]>([]);

  const [
    ativo,
    setAtivo,
  ] = useState(true);

  const [
    salvando,
    setSalvando,
  ] = useState(false);

  useEffect(() => {
    if (!aberto) {
      return;
    }

    const desligarPessoas =
      onValue(
        ref(db, "usuarios-v2"),
        (snapshot) => {
          const dados =
            snapshot.val();

          const lista: Pessoa[] =
            dados
              ? Object.entries(dados)
                  .map(
                    ([id, valor]) => ({
                      id,
                      ...(valor as Omit<
                        Pessoa,
                        "id"
                      >),
                    })
                  )
                  .sort(
                    (a, b) =>
                      String(
                        a.nome ||
                        a.email ||
                        ""
                      ).localeCompare(
                        String(
                          b.nome ||
                          b.email ||
                          ""
                        ),
                        "pt-BR"
                      )
                  )
              : [];

          setPessoas(lista);
        }
      );

    const desligarLocais =
      onValue(
        ref(db, "locais-v2"),
        (snapshot) => {
          const dados =
            snapshot.val();

          const lista: Local[] =
            dados
              ? Object.entries(dados)
                  .map(
                    ([id, valor]) => ({
                      id,
                      ...(valor as Omit<
                        Local,
                        "id"
                      >),
                    })
                  )
                  .sort(
                    (a, b) =>
                      String(
                        a.nome || ""
                      ).localeCompare(
                        String(
                          b.nome || ""
                        ),
                        "pt-BR"
                      )
                  )
              : [];

          setLocais(lista);
        }
      );

    const desligarUnidades =
      onValue(
        ref(db, "unidades-v2"),
        (snapshot) => {
          const dados =
            snapshot.val();

          setUnidades(
            dados
              ? Object.entries(dados)
                  .map(
                    ([id, valor]) => ({
                      id,
                      ...(valor as Omit<
                        Unidade,
                        "id"
                      >),
                    })
                  )
              : []
          );
        }
      );

    return () => {
      desligarPessoas();
      desligarLocais();
      desligarUnidades();
    };
  }, [aberto]);

  const pessoaSelecionada =
    pessoas.find(
      (pessoa) =>
        pessoa.id === pessoaId
    );

  const localSelecionado =
    locais.find(
      (local) =>
        local.id === localId
    );

  const entradaVinculoExistente =
    useMemo(() => {
      if (
        !pessoaSelecionada ||
        !localSelecionado
      ) {
        return null;
      }

      function procurar(
        origem:
          Record<string, unknown> | undefined
      ) {
        if (!origem) {
        return null;
      }

      if (!localSelecionado) {
        return null;
      }

        const encontrado =
          (Object.entries(origem) as Array<[string, VinculoExistente]>)
        .find(
            ([
              chave,
              dados,
            ]) =>
              chave ===
                localSelecionado.id ||
              chave ===
                localSelecionado.slug ||
              dados.localId ===
                localSelecionado.id ||
              (
                Boolean(
                  localSelecionado.slug
                ) &&
                dados.localSlug ===
                  localSelecionado.slug
              )
          );

        if (!encontrado) {
          return null;
        }

        return {
          chave:
            encontrado[0],

          dados:
            encontrado[1],
        };
      }

      return (
        procurar(
          pessoaSelecionada.locais
        ) ||
        procurar(
          pessoaSelecionada.condominios
        )
      );
    }, [
      pessoaSelecionada,
      localSelecionado,
    ]);

  const vinculoExistente =
    entradaVinculoExistente
      ?.dados ||
    null;

  const chaveVinculoExistente =
    entradaVinculoExistente
      ?.chave ||
    "";

  const jaExiste =
    Boolean(
      vinculoExistente
    );

  useEffect(() => {
    if (
      !pessoaId ||
      !localId
    ) {
      return;
    }

    if (
      !vinculoExistente
    ) {
      setPerfis([]);
      setPerfilPadrao("");
      setUnidadesSelecionadas([]);
      setAtivo(true);

      return;
    }

    const perfisAtivos =
      Object.entries(
        vinculoExistente.perfis ||
        {}
      )
        .filter(
          ([, valor]) =>
            valor === true
        )
        .map(
          ([perfil]) =>
            perfil
        );

    const perfilPrincipal =
      String(
        vinculoExistente
          .perfilPrincipal ||
        ""
      ).trim();

    if (
      perfilPrincipal &&
      !perfisAtivos.includes(
        perfilPrincipal
      )
    ) {
      perfisAtivos.push(
        perfilPrincipal
      );
    }

    const unidadesAtivas =
      Object.entries(
        vinculoExistente.unidades ||
        {}
      )
        .filter(
          ([, valor]) =>
            valor === true
        )
        .map(
          ([unidadeId]) =>
            unidadeId
        );

    setPerfis(
      perfisAtivos
    );

    setPerfilPadrao(
      perfilPrincipal ||
      perfisAtivos[0] ||
      ""
    );

    setUnidadesSelecionadas(
      unidadesAtivas
    );

    setAtivo(
      vinculoExistente.ativo !==
        false
    );
  }, [
    pessoaId,
    localId,
    vinculoExistente,
  ]);
  const unidadesDoLocal =
    useMemo(
      () =>
        unidades
          .filter(
            (unidade) =>
              unidade.localId ===
                localId ||
              unidade.condominioId ===
                localId
          )
          .sort(
            (a, b) =>
              nomeUnidade(a).localeCompare(
                nomeUnidade(b),
                "pt-BR"
              )
          ),
      [
        unidades,
        localId,
      ]
    );

  function limpar() {
    setPessoaId("");
    setLocalId("");
    setPerfis([]);
    setPerfilPadrao("");
    setUnidadesSelecionadas([]);
    setAtivo(true);
  }

  function fechar() {
    if (salvando) {
      return;
    }

    limpar();
    onClose();
  }

  function alternarPerfil(
    perfil: string
  ) {
    setPerfis(
      (atual) => {
        if (
          atual.includes(perfil)
        ) {
          const novos =
            atual.filter(
              (item) =>
                item !== perfil
            );

          if (
            perfilPadrao ===
            perfil
          ) {
            setPerfilPadrao(
              novos[0] || ""
            );
          }

          return novos;
        }

        const novos = [
          ...atual,
          perfil,
        ];

        if (
          !perfilPadrao
        ) {
          setPerfilPadrao(
            perfil
          );
        }

        return novos;
      }
    );
  }

  function alternarUnidade(
    unidadeId: string
  ) {
    setUnidadesSelecionadas(
      (atual) =>
        atual.includes(
          unidadeId
        )
          ? atual.filter(
              (id) =>
                id !== unidadeId
            )
          : [
              ...atual,
              unidadeId,
            ]
    );
  }

  async function salvar() {
    if (
      !pessoaSelecionada ||
      !localSelecionado
    ) {
      alert(
        "Selecione a pessoa e o local."
      );

      return;
    }

    

    if (
      perfis.length ===
      0
    ) {
      alert(
        "Selecione pelo menos um perfil."
      );

      return;
    }

    if (
      !perfilPadrao ||
      !perfis.includes(
        perfilPadrao
      )
    ) {
      alert(
        "Escolha o perfil padrão."
      );

      return;
    }

    const agora =
      Date.now();

    const perfisObjeto:
      Record<
        string,
        boolean
      > = {};

    perfis.forEach(
      (perfil) => {
        perfisObjeto[
          perfil
        ] = true;
      }
    );

    const unidadesObjeto:
      Record<
        string,
        boolean
      > = {};

    unidadesSelecionadas.forEach(
      (unidadeId) => {
        unidadesObjeto[
          unidadeId
        ] = true;
      }
    );

    const dadosVinculo = {
      /*
       * Preserva dados que ja pertencem
       * ao vinculo, inclusive permissoes
       * operacionais e campos futuros.
       */
      ...(vinculoExistente ||
        {}),

      localId:
        localSelecionado.id,

      localNome:
        localSelecionado.nome ||
        localSelecionado.id,

      localSlug:
        localSelecionado.slug ||
        "",

      tipoLocal:
        localSelecionado.tipoLocal ||
        "outro",

      perfilPrincipal:
        perfilPadrao,

      perfis:
        perfisObjeto,

      unidades:
        unidadesObjeto,

      ativo,

      criadoEm:
        typeof vinculoExistente
          ?.criadoEm ===
        "number"
          ? vinculoExistente
              .criadoEm
          : agora,

      atualizadoEm:
        agora,
    };

    try {
      setSalvando(true);

      await update(
        ref(db),
        {
          [`usuarios-v2/${pessoaSelecionada.id}/locais/${localSelecionado.id}`]:
            dadosVinculo,

          [`usuarios-v2/${pessoaSelecionada.id}/condominios/${localSelecionado.id}`]:
            dadosVinculo,

          [`usuarios-v2/${pessoaSelecionada.id}/atualizadoEm`]:
            agora,

          [`vinculos-locais-v2/${localSelecionado.id}/${pessoaSelecionada.id}`]:
            {
              uid:
                pessoaSelecionada.id,

              nome:
                pessoaSelecionada.nome ||
                "",

              email:
                pessoaSelecionada.email ||
                "",

              ...dadosVinculo,
            },
        }
      );

      alert(
        "Novo vínculo criado com sucesso."
      );

      limpar();
      onClose();
    } catch (
      erro
    ) {
      console.error(
        "Erro ao criar vínculo:",
        erro
      );

      alert(
        "Não foi possível criar o vínculo."
      );
    } finally {
      setSalvando(false);
    }
  }

  if (!aberto) {
    return null;
  }

  return (
    <div className="fixed inset-0 z-[220] flex items-center justify-center bg-black/80 p-3">
      <div className="max-h-[94vh] w-full max-w-2xl overflow-y-auto rounded-3xl border border-violet-700 bg-slate-900 p-5">
        <div className="flex items-start justify-between">
          <div>
            <p className="text-xs font-black text-violet-300">
              ＋ NOVO VÍNCULO
            </p>

            <h3 className="mt-2 text-2xl font-black">
              Vincular pessoa a um local
            </h3>

            <p className="mt-2 text-sm text-slate-400">
              Nenhum novo usuário será criado.
            </p>
          </div>

          <button
            type="button"
            onClick={fechar}
            disabled={salvando}
            className="h-11 w-11 rounded-xl bg-slate-800 font-black"
          >
            ✕
          </button>
        </div>

        <div className="mt-6">
          <label className="text-xs font-black text-slate-400">
            PESSOA
          </label>

          <select
            value={pessoaId}
            onChange={(evento) => {
              setPessoaId(
                evento.target.value
              );
            }}
            className="mt-2 w-full rounded-xl border border-slate-700 bg-slate-800 p-3"
          >
            <option value="">
              Selecione a pessoa
            </option>

            {pessoas.map(
              (pessoa) => (
                <option
                  key={pessoa.id}
                  value={pessoa.id}
                >
                  {pessoa.nome ||
                    pessoa.email ||
                    pessoa.id}
                </option>
              )
            )}
          </select>
        </div>

        <div className="mt-5">
          <label className="text-xs font-black text-slate-400">
            LOCAL
          </label>

          <select
            value={localId}
            onChange={(evento) => {
              setLocalId(
                evento.target.value
              );

              setUnidadesSelecionadas(
                []
              );
            }}
            className="mt-2 w-full rounded-xl border border-slate-700 bg-slate-800 p-3"
          >
            <option value="">
              Selecione o local
            </option>

            {locais.map(
              (local) => (
                <option
                  key={local.id}
                  value={local.id}
                >
                  {local.nome ||
                    local.id}
                </option>
              )
            )}
          </select>
        </div>

        {jaExiste && (
          <div className="mt-4 rounded-xl border border-amber-700 bg-amber-950/30 p-4 font-bold text-amber-300">
            Esta pessoa já possui vínculo com o local selecionado.
          </div>
        )}

        <section className="mt-6">
          <p className="text-xs font-black text-slate-400">
            PERFIS
          </p>

          <div className="mt-3 grid gap-2 sm:grid-cols-2">
            {PERFIS.map(
              (perfil) => (
                <label
                  key={perfil}
                  className="flex cursor-pointer items-center gap-3 rounded-xl border border-slate-700 bg-slate-800 p-3"
                >
                  <input
                    type="checkbox"
                    checked={
                      perfis.includes(
                        perfil
                      )
                    }
                    onChange={() =>
                      alternarPerfil(
                        perfil
                      )
                    }
                    className="h-5 w-5"
                  />

                  <span className="font-bold">
                    {nomePerfil(
                      perfil
                    )}
                  </span>
                </label>
              )
            )}
          </div>
        </section>

        {perfis.length > 0 && (
          <section className="mt-6">
            <p className="text-xs font-black text-slate-400">
              PERFIL PADRÃO AO ENTRAR
            </p>

            <div className="mt-3 space-y-2">
              {perfis.map(
                (perfil) => (
                  <label
                    key={perfil}
                    className="flex cursor-pointer items-center gap-3 rounded-xl border border-blue-900 bg-blue-950/20 p-3"
                  >
                    <input
                      type="radio"
                      name="novo-vinculo-perfil"
                      checked={
                        perfilPadrao ===
                        perfil
                      }
                      onChange={() =>
                        setPerfilPadrao(
                          perfil
                        )
                      }
                      className="h-5 w-5"
                    />

                    <span className="font-bold">
                      {nomePerfil(
                        perfil
                      )}
                    </span>
                  </label>
                )
              )}
            </div>
          </section>
        )}

        <section className="mt-6">
          <p className="text-xs font-black text-slate-400">
            STATUS
          </p>

          <div className="mt-3 grid grid-cols-2 gap-3">
            <button
              type="button"
              onClick={() =>
                setAtivo(true)
              }
              className={`rounded-xl border p-3 font-black ${
                ativo
                  ? "border-green-500 bg-green-950 text-green-300"
                  : "border-slate-700 bg-slate-800"
              }`}
            >
              🟢 Ativo
            </button>

            <button
              type="button"
              onClick={() =>
                setAtivo(false)
              }
              className={`rounded-xl border p-3 font-black ${
                !ativo
                  ? "border-red-500 bg-red-950 text-red-300"
                  : "border-slate-700 bg-slate-800"
              }`}
            >
              🔴 Inativo
            </button>
          </div>
        </section>

        {localId && (
          <section className="mt-6">
            <p className="text-xs font-black text-slate-400">
              UNIDADES
            </p>

            {unidadesDoLocal.length ===
            0 ? (
              <div className="mt-3 rounded-xl border border-slate-700 bg-slate-800 p-4 text-sm text-slate-400">
                Este local não possui unidades cadastradas.
              </div>
            ) : (
              <div className="mt-3 max-h-64 space-y-2 overflow-y-auto rounded-xl border border-slate-700 bg-slate-950 p-3">
                {unidadesDoLocal.map(
                  (unidade) => (
                    <label
                      key={unidade.id}
                      className="flex cursor-pointer items-center gap-3 rounded-xl border border-slate-800 bg-slate-900 p-3"
                    >
                      <input
                        type="checkbox"
                        checked={
                          unidadesSelecionadas.includes(
                            unidade.id
                          )
                        }
                        onChange={() =>
                          alternarUnidade(
                            unidade.id
                          )
                        }
                        className="h-5 w-5"
                      />

                      <span className="font-bold">
                        {nomeUnidade(
                          unidade
                        )}
                      </span>
                    </label>
                  )
                )}
              </div>
            )}
          </section>
        )}

        <div className="mt-6 grid grid-cols-2 gap-3">
          <button
            type="button"
            onClick={fechar}
            disabled={salvando}
            className="rounded-xl bg-slate-700 py-3 font-black"
          >
            Cancelar
          </button>

          <button
            type="button"
            onClick={salvar}
            disabled={salvando}
            className="rounded-xl bg-green-600 py-3 font-black text-white disabled:cursor-not-allowed disabled:opacity-40"
          >
            {salvando
              ? "Salvando..."
              : jaExiste
                ? "💾 Atualizar vínculo"
                : "🔗 Criar vínculo"}
          </button>
        </div>
      </div>
    </div>
  );
}
