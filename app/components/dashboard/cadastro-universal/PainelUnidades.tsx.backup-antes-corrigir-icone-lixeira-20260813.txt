"use client";

import {
  useEffect,
  useMemo,
  useState,
} from "react";

import {
  onValue,
  ref,
  update,
} from "firebase/database";

import {
  db,
} from "../../../services/firebase";

type UnidadeBanco = {
  id?: string;
  unidadeId?: string;
  nome?: string;
  identificacao?: string;
  codigo?: string;
  numero?: string;
  slug?: string;

  localId?: string;
  localNome?: string;
  localSlug?: string;

  condominioId?: string;

  estruturaPaiId?: string;
  estruturaPaiNome?: string;
  bloco?: string;

  tipo?: string;
  tipoLocal?: string;

  status?: string;
  ativo?: boolean;
};

type LocalBanco = {
  id?: string;
  localId?: string;
  nome?: string;
  nomeExibicao?: string;
  slug?: string;
  tipo?: string;
  tipoLocal?: string;
  segmento?: string;

  unidades?: Record<
    string,
    UnidadeBanco
  >;

  estruturas?: {
    unidades?: Record<
      string,
      UnidadeBanco
    >;
  };
};

type VinculoUsuario = {
  localId?: string;
  localNome?: string;
  localSlug?: string;

  ativo?: boolean;

  unidades?: Record<
    string,
    boolean
  >;
};

type UsuarioBanco = {
  locais?: Record<
    string,
    VinculoUsuario
  >;

  condominios?: Record<
    string,
    VinculoUsuario
  >;

  vinculos?: Record<
    string,
    VinculoUsuario
  >;
};

type UnidadeResumo = {
  id: string;
  nome: string;
  tipo: string;
  status: string;

  localId: string;
  localNome: string;

  estrutura: string;

  totalPessoas: number;
};

function nomeTipo(
  tipo?: string
): string {
  const valor =
    String(
      tipo || ""
    ).toLowerCase();

  switch (valor) {
    case "apartamento":
      return "Apartamento";

    case "casa":
      return "Casa";

    case "residencia":
      return "Residência";

    case "sala":
      return "Sala";

    case "loja":
      return "Loja";

    default:
      return tipo || "Unidade";
  }
}

function unidadeAtiva(
  unidade: UnidadeBanco
): boolean {
  if (
    unidade.ativo === false
  ) {
    return false;
  }

  const status =
    String(
      unidade.status || ""
    ).toLowerCase();

  return ![
    "inativa",
    "inativo",
    "arquivada",
    "arquivado",
    "excluida",
    "excluido",
  ].includes(
    status
  );
}

function usuarioTemUnidade(
  usuario: UsuarioBanco,
  localId: string,
  unidadeId: string
): boolean {
  const fontes = [
    usuario.locais,
    usuario.condominios,
    usuario.vinculos,
  ];

  return fontes.some(
    (
      fonte
    ) => {
      if (!fonte) {
        return false;
      }

      return Object.entries(
        fonte
      ).some(
        (
          [
            vinculoId,
            vinculo,
          ]
        ) => {
          if (
            !vinculo ||
            vinculo.ativo ===
              false
          ) {
            return false;
          }

          const mesmoLocal =
            vinculoId ===
              localId ||
            vinculo.localId ===
              localId ||
            vinculo.localSlug ===
              localId;

          if (!mesmoLocal) {
            return false;
          }

          return (
            vinculo.unidades?.[
              unidadeId
            ] === true
          );
        }
      );
    }
  );
}

type Props = {
  onVerPessoasDaUnidade: (
    localId: string,
    unidadeId: string,
    unidadeNome: string
  ) => void;
};

export default function PainelUnidades({
  onVerPessoasDaUnidade,
}: Props) {
  const [
    locais,
    setLocais,
  ] = useState<
    Record<
      string,
      LocalBanco
    >
  >({});

  const [
    unidadesBanco,
    setUnidadesBanco,
  ] = useState<
    Record<
      string,
      UnidadeBanco
    >
  >({});

  const [
    usuarios,
    setUsuarios,
  ] = useState<
    Record<
      string,
      UsuarioBanco
    >
  >({});

  const [
    carregandoLocais,
    setCarregandoLocais,
  ] = useState(true);

  const [
    carregandoUnidades,
    setCarregandoUnidades,
  ] = useState(true);

  const [
    carregandoUsuarios,
    setCarregandoUsuarios,
  ] = useState(true);

  const [
    pesquisa,
    setPesquisa,
  ] = useState("");

  const [
    filtroLocal,
    setFiltroLocal,
  ] = useState("todos");

  const [
    filtroStatusUnidade,
    setFiltroStatusUnidade,
  ] = useState("ativas");

  const [
    arquivandoUnidadeId,
    setArquivandoUnidadeId,
  ] = useState<string | null>(
    null
  );

  useEffect(() => {
    const parar =
      onValue(
        ref(
          db,
          "locais-v2"
        ),
        (
          snapshot
        ) => {
          setLocais(
            snapshot.val() ||
            {}
          );

          setCarregandoLocais(
            false
          );
        }
      );

    return () => {
      parar();
    };
  }, []);

  useEffect(() => {
    const parar =
      onValue(
        ref(
          db,
          "unidades-v2"
        ),
        (
          snapshot
        ) => {
          setUnidadesBanco(
            snapshot.val() ||
            {}
          );

          setCarregandoUnidades(
            false
          );
        }
      );

    return () => {
      parar();
    };
  }, []);

  useEffect(() => {
    const parar =
      onValue(
        ref(
          db,
          "usuarios-v2"
        ),
        (
          snapshot
        ) => {
          setUsuarios(
            snapshot.val() ||
            {}
          );

          setCarregandoUsuarios(
            false
          );
        }
      );

    return () => {
      parar();
    };
  }, []);

  async function arquivarUnidade(
    unidade: UnidadeResumo
  ) {
    if (
      arquivandoUnidadeId
    ) {
      return;
    }

    const confirmado =
      window.confirm(
        'Arquivar "' +
          unidade.nome +
          '" de "' +
          unidade.localNome +
          '"?\n\n' +
          'A unidade saira da lista ativa.\n' +
          'Pessoas, vinculos e historico serao preservados.'
      );

    if (
      !confirmado
    ) {
      return;
    }

    try {
      setArquivandoUnidadeId(
        unidade.id
      );

      const agora =
        Date.now();

      const atualizacoes: Record<
        string,
        unknown
      > = {};

      const local =
        locais[
          unidade.localId
        ];

      // unidades-v2
      if (
        unidadesBanco[
          unidade.id
        ]
      ) {
        atualizacoes[
          "unidades-v2/" +
            unidade.id +
            "/status"
        ] = "arquivada";

        atualizacoes[
          "unidades-v2/" +
            unidade.id +
            "/ativo"
        ] = false;

        atualizacoes[
          "unidades-v2/" +
            unidade.id +
            "/atualizadoEm"
        ] = agora;
      }

      // Unidade diretamente dentro de locais-v2
      if (
        local?.unidades?.[
          unidade.id
        ]
      ) {
        atualizacoes[
          "locais-v2/" +
            unidade.localId +
            "/unidades/" +
            unidade.id +
            "/status"
        ] = "arquivada";

        atualizacoes[
          "locais-v2/" +
            unidade.localId +
            "/unidades/" +
            unidade.id +
            "/ativo"
        ] = false;

        atualizacoes[
          "locais-v2/" +
            unidade.localId +
            "/unidades/" +
            unidade.id +
            "/atualizadoEm"
        ] = agora;
      }

      // Unidade dentro de estruturas/unidades
      if (
        local?.estruturas
          ?.unidades?.[
            unidade.id
          ]
      ) {
        atualizacoes[
          "locais-v2/" +
            unidade.localId +
            "/estruturas/unidades/" +
            unidade.id +
            "/status"
        ] = "arquivada";

        atualizacoes[
          "locais-v2/" +
            unidade.localId +
            "/estruturas/unidades/" +
            unidade.id +
            "/ativo"
        ] = false;

        atualizacoes[
          "locais-v2/" +
            unidade.localId +
            "/estruturas/unidades/" +
            unidade.id +
            "/atualizadoEm"
        ] = agora;
      }

      if (
        Object.keys(
          atualizacoes
        ).length === 0
      ) {
        throw new Error(
          "Registro oficial da unidade nao localizado."
        );
      }

      await update(
        ref(
          db
        ),
        atualizacoes
      );
    } catch (
      erro
    ) {
      console.error(
        "Erro ao arquivar unidade:",
        erro
      );

      window.alert(
        "Nao foi possivel arquivar a unidade."
      );
    } finally {
      setArquivandoUnidadeId(
        null
      );
    }
  }

  const unidades =
    useMemo<
      UnidadeResumo[]
    >(
      () => {
        const mapa =
          new Map<
            string,
            {
              unidade:
                UnidadeBanco;
              localId:
                string;
              localNome:
                string;
            }
          >();

        // ====================================================
        // 1. UNIDADES QUE JA ESTAO DENTRO DOS LOCAIS OFICIAIS
        // ====================================================

        Object.entries(
          locais
        ).forEach(
          (
            [
              localId,
              local,
            ]
          ) => {
            const localNome =
              local.nomeExibicao ||
              local.nome ||
              localId;

            const unidadesDiretas =
              local.unidades ||
              {};

            Object.entries(
              unidadesDiretas
            ).forEach(
              (
                [
                  unidadeId,
                  unidade,
                ]
              ) => {
                mapa.set(
                  unidadeId,
                  {
                    unidade: {
                      ...unidade,
                      id:
                        unidade.id ||
                        unidadeId,
                    },
                    localId,
                    localNome,
                  }
                );
              }
            );

            const unidadesEstruturas =
              local.estruturas
                ?.unidades ||
              {};

            Object.entries(
              unidadesEstruturas
            ).forEach(
              (
                [
                  unidadeId,
                  unidade,
                ]
              ) => {
                if (
                  !mapa.has(
                    unidadeId
                  )
                ) {
                  mapa.set(
                    unidadeId,
                    {
                      unidade: {
                        ...unidade,
                        id:
                          unidade.id ||
                          unidadeId,
                      },
                      localId,
                      localNome,
                    }
                  );
                }
              }
            );
          }
        );

        // ====================================================
        // 2. COMPLEMENTO DE unidades-v2
        // SOMENTE SE TIVER localId OFICIAL
        // ====================================================

        Object.entries(
          unidadesBanco
        ).forEach(
          (
            [
              unidadeId,
              unidade,
            ]
          ) => {
            const localId =
              unidade.localId ||
              unidade.condominioId ||
              "";

            if (
              !localId ||
              !locais[localId]
            ) {
              return;
            }

            if (
              mapa.has(
                unidadeId
              )
            ) {
              return;
            }

            const local =
              locais[localId];

            mapa.set(
              unidadeId,
              {
                unidade: {
                  ...unidade,
                  id:
                    unidade.id ||
                    unidadeId,
                },
                localId,
                localNome:
                  unidade.localNome ||
                  local.nomeExibicao ||
                  local.nome ||
                  localId,
              }
            );
          }
        );

        // ====================================================
        // 3. NORMALIZAR + CONTAR PESSOAS
        // ====================================================

        return Array.from(
          mapa.entries()
        )
          .map(
            (
              [
                unidadeId,
                dados,
              ]
            ) => {
              const unidade =
                dados.unidade;

              const totalPessoas =
                Object.values(
                  usuarios
                ).filter(
                  (
                    usuario
                  ) =>
                    usuarioTemUnidade(
                      usuario,
                      dados.localId,
                      unidadeId
                    )
                ).length;

              return {
                id:
                  unidadeId,

                nome:
                  unidade.nome ||
                  unidade.identificacao ||
                  unidade.codigo ||
                  unidade.numero ||
                  unidadeId,

                tipo:
                  nomeTipo(
                    unidade.tipo
                  ),

                status:
                  unidadeAtiva(
                    unidade
                  )
                    ? "ativa"
                    : "inativa",

                localId:
                  dados.localId,

                localNome:
                  dados.localNome,

                estrutura:
                  unidade.estruturaPaiNome ||
                  unidade.bloco ||
                  "",

                totalPessoas,
              };
            }
          )
          .sort(
            (
              a,
              b
            ) => {
              const local =
                a.localNome.localeCompare(
                  b.localNome,
                  "pt-BR"
                );

              if (
                local !== 0
              ) {
                return local;
              }

              return a.nome.localeCompare(
                b.nome,
                "pt-BR",
                {
                  numeric:
                    true,
                }
              );
            }
          );
      },
      [
        locais,
        unidadesBanco,
        usuarios,
      ]
    );

  const locaisDisponiveis =
    useMemo(
      () =>
        Object.entries(
          locais
        )
          .map(
            (
              [
                id,
                local,
              ]
            ) => ({
              id,

              nome:
                local.nomeExibicao ||
                local.nome ||
                id,
            })
          )
          .sort(
            (
              a,
              b
            ) =>
              a.nome.localeCompare(
                b.nome,
                "pt-BR"
              )
          ),
      [
        locais,
      ]
    );

  const unidadesFiltradas =
    useMemo(
      () => {
        const termo =
          pesquisa
            .trim()
            .toLowerCase();

        return unidades.filter(
          (
            unidade
          ) => {
            const passaLocal =
              filtroLocal ===
                "todos" ||
              unidade.localId ===
                filtroLocal;

            const passaStatusUnidade =
              filtroStatusUnidade ===
                "todas" ||
              (
                filtroStatusUnidade ===
                  "ativas" &&
                unidade.status ===
                  "ativa"
              ) ||
              (
                filtroStatusUnidade ===
                  "arquivadas" &&
                unidade.status !==
                  "ativa"
              );

            const passaBusca =
              !termo ||
              [
                unidade.nome,
                unidade.tipo,
                unidade.localNome,
                unidade.estrutura,
                unidade.id,
              ]
                .join(" ")
                .toLowerCase()
                .includes(
                  termo
                );

            return (
              passaLocal &&
              passaStatusUnidade &&
              passaBusca
            );
          }
        );
      },
      [
        unidades,
        filtroLocal,
        filtroStatusUnidade,
        pesquisa,
      ]
    );

  const totalComPessoas =
    unidades.filter(
      (
        unidade
      ) =>
        unidade.totalPessoas >
        0
    ).length;

  const carregando =
    carregandoLocais ||
    carregandoUnidades ||
    carregandoUsuarios;

  return (
    <div className="space-y-5">
      <section className="rounded-3xl border border-violet-800 bg-gradient-to-r from-violet-950/50 to-slate-900 p-5 md:p-6">
        <p className="text-xs font-black uppercase tracking-wider text-violet-300">
          PAINEL DE UNIDADES
        </p>

        <h3 className="mt-2 text-2xl font-black text-white md:text-3xl">
          Unidades conectadas aos locais
        </h3>

        <p className="mt-2 max-w-3xl text-sm leading-relaxed text-slate-400">
          Visão universal de apartamentos, casas e demais unidades, conectando local e pessoas por meio dos vínculos do QR Core.
        </p>
      </section>

      <section className="grid grid-cols-2 gap-3 lg:grid-cols-3">
        <div className="rounded-2xl border border-slate-700 bg-slate-900 p-4">
          <p className="text-xs font-bold text-slate-400">
            UNIDADES
          </p>

          <p className="mt-2 text-3xl font-black text-white">
            {unidades.length}
          </p>
        </div>

        <div className="rounded-2xl border border-violet-800 bg-violet-950/20 p-4">
          <p className="text-xs font-bold text-violet-300">
            LOCAIS
          </p>

          <p className="mt-2 text-3xl font-black text-white">
            {
              locaisDisponiveis.length
            }
          </p>
        </div>

        <div className="col-span-2 rounded-2xl border border-blue-800 bg-blue-950/20 p-4 lg:col-span-1">
          <p className="text-xs font-bold text-blue-300">
            COM PESSOAS
          </p>

          <p className="mt-2 text-3xl font-black text-white">
            {totalComPessoas}
          </p>
        </div>
      </section>

      <section className="rounded-2xl border border-slate-800 bg-slate-900 p-4 md:p-5">
        <div className="grid gap-3 lg:grid-cols-[1fr_240px_200px]">
          <input
            value={
              pesquisa
            }
            onChange={(
              event
            ) =>
              setPesquisa(
                event.target.value
              )
            }
            placeholder="Pesquisar unidade, bloco, local ou tipo..."
            className="w-full rounded-xl border border-slate-700 bg-slate-800 p-3 text-white outline-none placeholder:text-slate-500 focus:border-violet-500"
          />

          <select
            value={
              filtroLocal
            }
            onChange={(
              event
            ) =>
              setFiltroLocal(
                event.target.value
              )
            }
            className="rounded-xl border border-slate-700 bg-slate-800 p-3 font-bold text-white outline-none focus:border-violet-500"
          >
            <option value="todos">
              Todos os locais
            </option>

            {locaisDisponiveis.map(
              (
                local
              ) => (
                <option
                  key={
                    local.id
                  }
                  value={
                    local.id
                  }
                >
                  {local.nome}
                </option>
              )
            )}
          </select>

          <select
            value={
              filtroStatusUnidade
            }
            onChange={(
              event
            ) =>
              setFiltroStatusUnidade(
                event.target.value
              )
            }
            className="rounded-xl border border-slate-700 bg-slate-800 p-3 font-bold text-white outline-none focus:border-violet-500"
          >
            <option value="ativas">
              Ativas
            </option>

            <option value="arquivadas">
              Arquivadas / inativas
            </option>

            <option value="todas">
              Todas
            </option>
          </select>
        </div>

        <p className="mt-3 text-sm text-slate-400">
          {
            unidadesFiltradas.length
          } unidade
          {
            unidadesFiltradas.length ===
            1
              ? ""
              : "s"
          } encontrada
          {
            unidadesFiltradas.length ===
            1
              ? ""
              : "s"
          }.
        </p>
      </section>

      {carregando ? (
        <section className="rounded-3xl border border-slate-800 bg-slate-900 p-8 text-center">
          <p className="font-black text-slate-300">
            Carregando unidades...
          </p>
        </section>
      ) : (
        <section className="grid gap-4 sm:grid-cols-2 xl:grid-cols-4">
          {unidadesFiltradas.map(
            (
              unidade
            ) => (
              <article
                key={
                  unidade.id
                }
                className="rounded-3xl border border-slate-700 bg-slate-900 p-5"
              >
                <div className="flex items-start justify-between gap-3">
                  <div className="min-w-0">
                    <p className="text-[10px] font-black uppercase tracking-wider text-violet-400">
                      {unidade.tipo}
                    </p>

                    <h4 className="mt-1 text-lg font-black text-white">
                      {unidade.nome}
                    </h4>
                  </div>

                  <span
                    className={`shrink-0 rounded-full px-3 py-1 text-[10px] font-black uppercase ${
                      unidade.status ===
                      "ativa"
                        ? "bg-green-950 text-green-300"
                        : "bg-slate-800 text-slate-400"
                    }`}
                  >
                    {
                      unidade.status
                    }
                  </span>

                  {unidade.status ===
                    "ativa" && (
                    <button
                      type="button"
                      title="Arquivar unidade"
                      onClick={() =>
                        void arquivarUnidade(
                          unidade
                        )
                      }
                      disabled={
                        arquivandoUnidadeId ===
                        unidade.id
                      }
                      className="flex h-8 w-8 shrink-0 items-center justify-center rounded-lg border border-red-900 bg-red-950/30 text-sm text-red-300 transition hover:bg-red-950/60 disabled:opacity-40"
                    >
                      {
                        arquivandoUnidadeId ===
                        unidade.id
                          ? "..."
                          : "??"
                      }
                    </button>
                  )}
                </div>

                <div className="mt-4 rounded-xl border border-slate-700 bg-slate-800 p-3">
                  <p className="text-[10px] font-black uppercase text-slate-500">
                    Local
                  </p>

                  <p className="mt-1 truncate text-sm font-black text-white">
                    {unidade.localNome}
                  </p>
                </div>

                {unidade.estrutura && (
                  <p className="mt-3 text-xs font-bold text-slate-400">
                    {unidade.estrutura}
                  </p>
                )}

                <button
                  type="button"
                  disabled={
                    unidade.totalPessoas === 0
                  }
                  onClick={() =>
                    onVerPessoasDaUnidade(
                      unidade.localId,
                      unidade.id,
                      unidade.nome
                    )
                  }
                  className="mt-4 w-full rounded-xl border border-blue-900 bg-blue-950/20 p-3 text-left transition hover:bg-blue-950/40 disabled:cursor-not-allowed disabled:opacity-60"
                >
                  <p className="text-[10px] font-black uppercase text-blue-400">
                    Pessoas vinculadas
                  </p>

                  <p className="mt-1 text-xl font-black text-white">
                    {
                      unidade.totalPessoas
                    }
                  </p>

                  <p className="mt-2 text-xs font-black text-blue-300">
                    {
                      unidade.totalPessoas > 0
                        ? `Ver ${unidade.totalPessoas} pessoa${unidade.totalPessoas === 1 ? "" : "s"}`
                        : "Nenhuma pessoa vinculada"
                    }
                  </p>
                </button>
              </article>
            )
          )}

          {unidadesFiltradas.length ===
            0 && (
            <div className="rounded-3xl border border-slate-800 bg-slate-900 p-8 text-center sm:col-span-2 xl:col-span-4">
              <p className="font-black text-slate-300">
                Nenhuma unidade encontrada.
              </p>
            </div>
          )}
        </section>
      )}
    </div>
  );
}
