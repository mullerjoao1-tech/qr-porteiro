"use client";

import {
  useEffect,
  useMemo,
  useState,
} from "react";

import {
  onValue,
  ref,
  update,
} from "firebase/database";

import {
  db,
} from "../../../services/firebase";

import NovoVinculoModal from "./NovoVinculoModal";

type VinculoUsuario = {
  localId?: string;
  localNome?: string;
  localSlug?: string;
  tipoLocal?: string;

  perfilPrincipal?: string;

  perfis?: Record<
    string,
    boolean | undefined
  >;

  unidades?: Record<
    string,
    boolean
  >;

  ativo?: boolean;
  criadoEm?: number;
  atualizadoEm?: number;
};

type PessoaUniversal = {
  id: string;
  nome?: string;
  email?: string;

  locais?: Record<
    string,
    VinculoUsuario
  >;

  condominios?: Record<
    string,
    VinculoUsuario
  >;
};

type LocalUniversal = {
  id: string;
  nome?: string;
  slug?: string;
  tipoLocal?: string;
};

type UnidadeUniversal = {
  id: string;
  localId?: string;
  condominioId?: string;
  bloco?: string;
  nome?: string;
  codigo?: string;
};

type VinculoNormalizado = {
  id: string;

  pessoaId: string;
  pessoaNome: string;
  pessoaEmail: string;

  localId: string;
  localNome: string;
  localSlug: string;
  tipoLocal: string;

  perfilPrincipal: string;
  perfis: string[];

  unidades: string[];

  ativo: boolean;

  criadoEm?: number;
  atualizadoEm?: number;
};

const PERFIS = [
  "morador",
  "sindico",
  "administradora",
  "gestor_local",
  "porteiro",
  "central",
  "funcionario",
  "financeiro",
  "prestador",
  "conselheiro",
  "administrador_master",
  "proprietario",
  "inquilino",
  "responsavel",
  "gerente",
  "outro",
];

function nomePerfil(
  valor?: string
) {
  const nomes: Record<
    string,
    string
  > = {
    morador: "Morador",
    sindico: "Síndico",
    administradora: "Administradora",
    gestor_local: "Gestor local",
    porteiro: "Porteiro",
    central: "Central",
    funcionario: "Funcionário",
    financeiro: "Financeiro",
    prestador: "Prestador",
    conselheiro: "Conselheiro",
    administrador_master:
      "Administrador master",
    proprietario: "Proprietário",
    inquilino: "Inquilino",
    responsavel: "Responsável",
    gerente: "Gerente",
    outro: "Outro",
  };

  return (
    nomes[valor || ""] ||
    valor ||
    "Não informado"
  );
}

function nomeTipoLocal(
  valor?: string
) {
  const nomes: Record<
    string,
    string
  > = {
    condominio: "Condomínio",
    residencia: "Residência",
    beauty: "Beauty",
    barbearia: "Barbearia",
    clinica: "Clínica",
    empresa: "Empresa",
    restaurante: "Restaurante",
    outro: "Outro",
  };

  return (
    nomes[valor || ""] ||
    valor ||
    "Local"
  );
}

function iconeTipoLocal(
  valor?: string
) {
  const icones: Record<
    string,
    string
  > = {
    condominio: "🏢",
    residencia: "🏠",
    beauty: "💇",
    barbearia: "💈",
    clinica: "🏥",
    empresa: "🏭",
    restaurante: "🍽️",
    outro: "📍",
  };

  return (
    icones[valor || ""] ||
    "📍"
  );
}

function nomeUnidade(
  unidade:
    UnidadeUniversal
) {
  const nome =
    unidade.nome ||
    unidade.codigo ||
    unidade.id;

  return unidade.bloco
    ? `${unidade.bloco} • ${nome}`
    : nome;
}

function obterVinculos(
  pessoa:
    PessoaUniversal
) {
  const locais =
    pessoa.locais || {};

  if (
    Object.keys(locais)
      .length >
    0
  ) {
    return Object.entries(
      locais
    );
  }

  return Object.entries(
    pessoa.condominios ||
      {}
  );
}

export default function PainelVinculos() {
  const [
    pessoas,
    setPessoas,
  ] =
    useState<
      PessoaUniversal[]
    >([]);

  const [
    locais,
    setLocais,
  ] =
    useState<
      LocalUniversal[]
    >([]);

  const [
    unidades,
    setUnidades,
  ] =
    useState<
      UnidadeUniversal[]
    >([]);

  const [
    busca,
    setBusca,
  ] =
    useState("");

  const [
    filtroStatus,
    setFiltroStatus,
  ] =
    useState("todos");

  const [
    filtroPerfil,
    setFiltroPerfil,
  ] =
    useState("todos");

  const [
    carregando,
    setCarregando,
  ] =
    useState(true);

  const [
    vinculoEditando,
    setVinculoEditando,
  ] =
    useState<
      VinculoNormalizado | null
    >(null);

  const [
    perfisEditando,
    setPerfisEditando,
  ] =
    useState<string[]>([]);

  const [
    perfilPrincipalEditando,
    setPerfilPrincipalEditando,
  ] =
    useState("");

  const [
    ativoEditando,
    setAtivoEditando,
  ] =
    useState(true);

  const [
    unidadesEditando,
    setUnidadesEditando,
  ] =
    useState<string[]>([]);

  const [
    salvando,
    setSalvando,
  ] =
    useState(false);

  const [
    excluindoVinculoId,
    setExcluindoVinculoId,
  ] =
    useState<string | null>(
      null
    );

  const [
    novoVinculoAberto,
    setNovoVinculoAberto,
  ] =
    useState(false);

  useEffect(() => {
    let pessoasProntas =
      false;

    let locaisProntos =
      false;

    let unidadesProntas =
      false;

    function conferir() {
      if (
        pessoasProntas &&
        locaisProntos &&
        unidadesProntas
      ) {
        setCarregando(
          false
        );
      }
    }

    const desligarPessoas =
      onValue(
        ref(
          db,
          "usuarios-v2"
        ),
        (
          snapshot
        ) => {
          const dados =
            snapshot.val();

          setPessoas(
            dados
              ? Object.entries(
                  dados
                ).map(
                  ([
                    id,
                    valor,
                  ]) => ({
                    id,
                    ...(valor as Omit<
                      PessoaUniversal,
                      "id"
                    >),
                  })
                )
              : []
          );

          pessoasProntas =
            true;

          conferir();
        }
      );

    const desligarLocais =
      onValue(
        ref(
          db,
          "locais-v2"
        ),
        (
          snapshot
        ) => {
          const dados =
            snapshot.val();

          setLocais(
            dados
              ? Object.entries(
                  dados
                ).map(
                  ([
                    id,
                    valor,
                  ]) => ({
                    id,
                    ...(valor as Omit<
                      LocalUniversal,
                      "id"
                    >),
                  })
                )
              : []
          );

          locaisProntos =
            true;

          conferir();
        }
      );

    const desligarUnidades =
      onValue(
        ref(
          db,
          "unidades-v2"
        ),
        (
          snapshot
        ) => {
          const dados =
            snapshot.val();

          setUnidades(
            dados
              ? Object.entries(
                  dados
                ).map(
                  ([
                    id,
                    valor,
                  ]) => ({
                    id,
                    ...(valor as Omit<
                      UnidadeUniversal,
                      "id"
                    >),
                  })
                )
              : []
          );

          unidadesProntas =
            true;

          conferir();
        }
      );

    return () => {
      desligarPessoas();
      desligarLocais();
      desligarUnidades();
    };
  }, []);

  const vinculos =
    useMemo(() => {
      const lista:
        VinculoNormalizado[] =
        [];

      pessoas.forEach(
        (
          pessoa
        ) => {
          obterVinculos(
            pessoa
          ).forEach(
            ([
              chaveLocal,
              vinculo,
            ]) => {
              const localId =
                vinculo.localId ||
                chaveLocal;

              const local =
                locais.find(
                  (
                    item
                  ) =>
                    item.id ===
                      localId ||
                    item.slug ===
                      vinculo.localSlug ||
                    item.slug ===
                      chaveLocal
                );

              const perfis =
                Object.entries(
                  vinculo.perfis ||
                    {}
                )
                  .filter(
                    ([
                      ,
                      ativo,
                    ]) =>
                      ativo
                  )
                  .map(
                    ([
                      perfil,
                    ]) =>
                      perfil
                  );

              if (
                vinculo.perfilPrincipal &&
                !perfis.includes(
                  vinculo.perfilPrincipal
                )
              ) {
                perfis.unshift(
                  vinculo.perfilPrincipal
                );
              }

              lista.push({
                id:
                  `${pessoa.id}-${localId}`,

                pessoaId:
                  pessoa.id,

                pessoaNome:
                  pessoa.nome ||
                  "Nome não informado",

                pessoaEmail:
                  pessoa.email ||
                  "E-mail não informado",

                localId,

                localNome:
                  vinculo.localNome ||
                  local?.nome ||
                  chaveLocal,

                localSlug:
                  vinculo.localSlug ||
                  local?.slug ||
                  "",

                tipoLocal:
                  vinculo.tipoLocal ||
                  local?.tipoLocal ||
                  "outro",

                perfilPrincipal:
                  vinculo.perfilPrincipal ||
                  perfis[0] ||
                  "outro",

                perfis:
                  perfis.length >
                  0
                    ? perfis
                    : [
                        "outro",
                      ],

                unidades:
                  Object.entries(
                    vinculo.unidades ||
                      {}
                  )
                    .filter(
                      ([
                        ,
                        ativo,
                      ]) =>
                        ativo
                    )
                    .map(
                      ([
                        unidadeId,
                      ]) =>
                        unidadeId
                    ),

                ativo:
                  vinculo.ativo !==
                  false,

                criadoEm:
                  vinculo.criadoEm,

                atualizadoEm:
                  vinculo.atualizadoEm,
              });
            }
          );
        }
      );

      return lista.sort(
        (
          a,
          b
        ) =>
          a.pessoaNome.localeCompare(
            b.pessoaNome,
            "pt-BR"
          )
      );
    }, [
      pessoas,
      locais,
    ]);

  const perfisDisponiveis =
    useMemo(
      () =>
        Array.from(
          new Set(
            vinculos.flatMap(
              (
                vinculo
              ) =>
                vinculo.perfis
            )
          )
        ).sort(),
      [
        vinculos,
      ]
    );

  const vinculosFiltrados =
    useMemo(() => {
      const termo =
        busca
          .trim()
          .toLowerCase();

      return vinculos.filter(
        (
          vinculo
        ) => {
          const passaBusca =
            !termo ||
            [
              vinculo.pessoaNome,
              vinculo.pessoaEmail,
              vinculo.localNome,
              vinculo.localSlug,
              vinculo.perfilPrincipal,
              ...vinculo.perfis,
            ]
              .filter(
                Boolean
              )
              .join(" ")
              .toLowerCase()
              .includes(
                termo
              );

          const passaStatus =
            filtroStatus ===
              "todos" ||
            (
              filtroStatus ===
                "ativo" &&
              vinculo.ativo
            ) ||
            (
              filtroStatus ===
                "inativo" &&
              !vinculo.ativo
            );

          const passaPerfil =
            filtroPerfil ===
              "todos" ||
            vinculo.perfis.includes(
              filtroPerfil
            );

          return (
            passaBusca &&
            passaStatus &&
            passaPerfil
          );
        }
      );
    }, [
      vinculos,
      busca,
      filtroStatus,
      filtroPerfil,
    ]);

  const unidadesDoLocal =
    useMemo(() => {
      if (
        !vinculoEditando
      ) {
        return [];
      }

      return unidades
        .filter(
          (
            unidade
          ) =>
            unidade.localId ===
              vinculoEditando.localId ||
            unidade.condominioId ===
              vinculoEditando.localId
        )
        .sort(
          (
            a,
            b
          ) =>
            nomeUnidade(
              a
            ).localeCompare(
              nomeUnidade(
                b
              ),
              "pt-BR"
            )
        );
    }, [
      unidades,
      vinculoEditando,
    ]);

  function abrirGerenciamento(
    vinculo:
      VinculoNormalizado
  ) {
    const perfis =
      vinculo.perfis.length >
      0
        ? vinculo.perfis
        : [
            vinculo.perfilPrincipal,
          ];

    setVinculoEditando(
      vinculo
    );

    setPerfisEditando(
      perfis
    );

    setPerfilPrincipalEditando(
      vinculo.perfilPrincipal ||
      perfis[0] ||
      "outro"
    );

    setAtivoEditando(
      vinculo.ativo
    );

    setUnidadesEditando(
      vinculo.unidades
    );
  }

  function alternarPerfil(
    perfil:
      string
  ) {
    setPerfisEditando(
      (
        atual
      ) => {
        if (
          atual.includes(
            perfil
          )
        ) {
          if (
            atual.length ===
            1
          ) {
            alert(
              "O vínculo precisa ter pelo menos um perfil."
            );

            return atual;
          }

          const novos =
            atual.filter(
              (
                item
              ) =>
                item !==
                perfil
            );

          if (
            perfil ===
            perfilPrincipalEditando
          ) {
            setPerfilPrincipalEditando(
              novos[0]
            );
          }

          return novos;
        }

        return [
          ...atual,
          perfil,
        ];
      }
    );
  }

  function alternarUnidade(
    unidadeId:
      string
  ) {
    setUnidadesEditando(
      (
        atual
      ) =>
        atual.includes(
          unidadeId
        )
          ? atual.filter(
              (
                id
              ) =>
                id !==
                unidadeId
            )
          : [
              ...atual,
              unidadeId,
            ]
    );
  }

  async function salvarVinculo() {
    if (
      !vinculoEditando ||
      salvando
    ) {
      return;
    }

    if (
      perfisEditando.length ===
      0
    ) {
      alert(
        "Selecione pelo menos um perfil."
      );

      return;
    }

    if (
      !perfisEditando.includes(
        perfilPrincipalEditando
      )
    ) {
      alert(
        "O perfil padrão precisa estar entre os perfis selecionados."
      );

      return;
    }

    const agora =
      Date.now();

    const perfisObjeto:
      Record<
        string,
        boolean
      > = {};

    perfisEditando.forEach(
      (
        perfil
      ) => {
        perfisObjeto[
          perfil
        ] =
          true;
      }
    );

    const unidadesObjeto:
      Record<
        string,
        boolean
      > = {};

    unidadesEditando.forEach(
      (
        unidadeId
      ) => {
        unidadesObjeto[
          unidadeId
        ] =
          true;
      }
    );

    const dadosVinculo = {
      localId:
        vinculoEditando.localId,

      localNome:
        vinculoEditando.localNome,

      localSlug:
        vinculoEditando.localSlug,

      tipoLocal:
        vinculoEditando.tipoLocal,

      perfilPrincipal:
        perfilPrincipalEditando,

      perfis:
        perfisObjeto,

      unidades:
        unidadesObjeto,

      ativo:
        ativoEditando,

      criadoEm:
        vinculoEditando.criadoEm ||
        agora,

      atualizadoEm:
        agora,
    };

    try {
      setSalvando(
        true
      );

      await update(
        ref(db),
        {
          [`usuarios-v2/${vinculoEditando.pessoaId}/locais/${vinculoEditando.localId}`]:
            dadosVinculo,

          [`usuarios-v2/${vinculoEditando.pessoaId}/condominios/${vinculoEditando.localId}`]:
            dadosVinculo,

          [`usuarios-v2/${vinculoEditando.pessoaId}/atualizadoEm`]:
            agora,
        }
      );

      setVinculoEditando(
        null
      );

      alert(
        "Vínculo atualizado com sucesso."
      );
    } catch (
      erro
    ) {
      console.error(
        "Erro ao salvar vínculo:",
        erro
      );

      alert(
        "Não foi possível salvar o vínculo."
      );
    } finally {
      setSalvando(
        false
      );
    }
  }

  const totalAtivos =
    vinculos.filter(
      (
        item
      ) =>
        item.ativo
    ).length;

  const pessoasVinculadas =
    new Set(
      vinculos.map(
        (
          item
        ) =>
          item.pessoaId
      )
    ).size;

  const multiplosLocais =
    pessoas.filter(
      (
        pessoa
      ) =>
        obterVinculos(
          pessoa
        ).length >
        1
    ).length;

  return (
    <div className="space-y-5">
      <section className="rounded-3xl border border-violet-800 bg-gradient-to-r from-violet-950/70 to-slate-900 p-5 md:p-6">
        <div className="flex flex-col gap-4 md:flex-row md:items-center md:justify-between">
          <div>
            <p className="text-xs font-black uppercase tracking-wider text-violet-300">
              🔗 VÍNCULOS DO QR CORE
            </p>

            <h3 className="mt-2 text-2xl font-black text-white md:text-3xl">
              Pessoa → Local → Perfis
            </h3>

            <p className="mt-2 text-sm text-slate-400">
              Uma pessoa pode possuir vários perfis no mesmo local, com um perfil padrão para entrada.
            </p>
          </div>

          <button
            type="button"
            onClick={() =>
              setNovoVinculoAberto(true)
            }
            className="rounded-xl bg-violet-600 px-5 py-3 font-black text-white"
          >
            ＋ Novo vínculo
          </button>
        </div>
      </section>

      <section className="grid grid-cols-2 gap-3 lg:grid-cols-4">
        <div className="rounded-2xl border border-slate-700 bg-slate-900 p-4">
          <p className="text-xs font-bold text-slate-400">
            TOTAL DE VÍNCULOS
          </p>

          <p className="mt-2 text-3xl font-black">
            {vinculos.length}
          </p>
        </div>

        <div className="rounded-2xl border border-green-800 bg-green-950/25 p-4">
          <p className="text-xs font-bold text-green-300">
            ATIVOS
          </p>

          <p className="mt-2 text-3xl font-black">
            {totalAtivos}
          </p>
        </div>

        <div className="rounded-2xl border border-blue-800 bg-blue-950/25 p-4">
          <p className="text-xs font-bold text-blue-300">
            PESSOAS VINCULADAS
          </p>

          <p className="mt-2 text-3xl font-black">
            {pessoasVinculadas}
          </p>
        </div>

        <div className="rounded-2xl border border-violet-800 bg-violet-950/25 p-4">
          <p className="text-xs font-bold text-violet-300">
            MÚLTIPLOS LOCAIS
          </p>

          <p className="mt-2 text-3xl font-black">
            {multiplosLocais}
          </p>
        </div>
      </section>

      <section className="rounded-2xl border border-slate-800 bg-slate-900 p-4">
        <div className="grid gap-3 lg:grid-cols-[1fr_210px_220px]">
          <input
            value={busca}
            onChange={(
              evento
            ) =>
              setBusca(
                evento.target.value
              )
            }
            placeholder="Pesquisar pessoa, local ou perfil..."
            className="rounded-xl border border-slate-700 bg-slate-800 p-3 outline-none"
          />

          <select
            value={filtroStatus}
            onChange={(
              evento
            ) =>
              setFiltroStatus(
                evento.target.value
              )
            }
            className="rounded-xl border border-slate-700 bg-slate-800 p-3"
          >
            <option value="todos">
              Todos os status
            </option>

            <option value="ativo">
              Ativos
            </option>

            <option value="inativo">
              Inativos
            </option>
          </select>

          <select
            value={filtroPerfil}
            onChange={(
              evento
            ) =>
              setFiltroPerfil(
                evento.target.value
              )
            }
            className="rounded-xl border border-slate-700 bg-slate-800 p-3"
          >
            <option value="todos">
              Todos os perfis
            </option>

            {perfisDisponiveis.map(
              (
                perfil
              ) => (
                <option
                  key={
                    perfil
                  }
                  value={
                    perfil
                  }
                >
                  {nomePerfil(
                    perfil
                  )}
                </option>
              )
            )}
          </select>
        </div>
      </section>

      {carregando ? (
        <div className="rounded-2xl border border-slate-700 bg-slate-900 p-8 text-center">
          Carregando vínculos...
        </div>
      ) : (
        <section className="grid gap-4 md:grid-cols-2 xl:grid-cols-3">
          {vinculosFiltrados.map(
            (
              vinculo
            ) => (
              <article
                key={
                  vinculo.id
                }
                className="rounded-2xl border border-slate-700 bg-slate-800 p-4"
              >
                <div className="flex items-start justify-between gap-3">
                  <span className="text-3xl">
                    👤
                  </span>

                  <div className="flex items-center gap-2">
                    <span
                      className={`rounded-full px-3 py-1 text-xs font-black ${
                        vinculo.ativo
                          ? "bg-green-950 text-green-300"
                          : "bg-red-950 text-red-300"
                      }`}
                    >
                      {vinculo.ativo
                        ? "ATIVO"
                        : "INATIVO"}
                    </span>

                    <button
                      type="button"
                      title="Excluir somente este vínculo"
                      onClick={() =>
                        void excluirVinculo(
                          vinculo
                        )
                      }
                      disabled={
                        excluindoVinculoId ===
                        vinculo.id
                      }
                      className="flex h-8 w-8 items-center justify-center rounded-lg border border-red-900 bg-red-950/30 text-sm transition hover:bg-red-900 disabled:opacity-40"
                    >
                      {excluindoVinculoId ===
                      vinculo.id
                        ? "…"
                        : "🗑"}
                    </button>
                  </div>
                </div>

                <h4 className="mt-4 text-xl font-black">
                  {vinculo.pessoaNome}
                </h4>

                <p className="text-xs text-slate-500">
                  {vinculo.pessoaEmail}
                </p>

                <div className="my-4 text-center text-xl">
                  ↓
                </div>

                <div className="rounded-xl border border-violet-800 bg-violet-950/20 p-3">
                  <p className="font-black">
                    {iconeTipoLocal(
                      vinculo.tipoLocal
                    )}{" "}
                    {vinculo.localNome}
                  </p>

                  <p className="mt-1 text-xs text-slate-500">
                    {nomeTipoLocal(
                      vinculo.tipoLocal
                    )}
                  </p>
                </div>

                <div className="mt-3">
                  <p className="text-xs font-black text-blue-300">
                    PERFIS
                  </p>

                  <div className="mt-2 flex flex-wrap gap-2">
                    {vinculo.perfis.map(
                      (
                        perfil
                      ) => (
                        <span
                          key={
                            perfil
                          }
                          className={`rounded-full border px-3 py-1 text-xs font-black ${
                            perfil ===
                            vinculo.perfilPrincipal
                              ? "border-blue-500 bg-blue-950 text-blue-300"
                              : "border-slate-600 bg-slate-900 text-slate-300"
                          }`}
                        >
                          {nomePerfil(
                            perfil
                          )}

                          {perfil ===
                            vinculo.perfilPrincipal &&
                            " ★"}
                        </span>
                      )
                    )}
                  </div>
                </div>

                <p className="mt-4 text-sm text-slate-400">
                  🚪{" "}
                  {
                    vinculo.unidades
                      .length
                  }{" "}
                  unidade(s)
                </p>

                <button
                  type="button"
                  onClick={() =>
                    abrirGerenciamento(
                      vinculo
                    )
                  }
                  className="mt-5 w-full rounded-xl bg-violet-600 px-4 py-3 font-black text-white"
                >
                  Gerenciar vínculo
                </button>
              </article>
            )
          )}
        </section>
      )}

      <NovoVinculoModal
        aberto={novoVinculoAberto}
        onClose={() =>
          setNovoVinculoAberto(false)
        }
      />

      {vinculoEditando && (
        <div className="fixed inset-0 z-[210] flex items-center justify-center bg-black/80 p-3">
          <div className="max-h-[94vh] w-full max-w-2xl overflow-y-auto rounded-3xl border border-violet-700 bg-slate-900 p-5">
            <div className="flex items-start justify-between">
              <div>
                <p className="text-xs font-black text-violet-300">
                  🔗 GERENCIAR VÍNCULO
                </p>

                <h3 className="mt-2 text-2xl font-black">
                  {vinculoEditando.pessoaNome}
                </h3>

                <p className="mt-1 text-slate-400">
                  {vinculoEditando.localNome}
                </p>
              </div>

              <button
                type="button"
                onClick={() =>
                  !salvando &&
                  setVinculoEditando(
                    null
                  )
                }
                className="h-11 w-11 rounded-xl bg-slate-800 font-black"
              >
                ✕
              </button>
            </div>

            <section className="mt-6">
              <p className="text-xs font-black text-slate-400">
                PERFIS
              </p>

              <div className="mt-3 grid gap-2 sm:grid-cols-2">
                {PERFIS.map(
                  (
                    perfil
                  ) => (
                    <label
                      key={
                        perfil
                      }
                      className="flex cursor-pointer items-center gap-3 rounded-xl border border-slate-700 bg-slate-800 p-3"
                    >
                      <input
                        type="checkbox"
                        checked={
                          perfisEditando.includes(
                            perfil
                          )
                        }
                        onChange={() =>
                          alternarPerfil(
                            perfil
                          )
                        }
                        className="h-5 w-5"
                      />

                      <span className="font-bold">
                        {nomePerfil(
                          perfil
                        )}
                      </span>
                    </label>
                  )
                )}
              </div>
            </section>

            <section className="mt-6">
              <p className="text-xs font-black text-slate-400">
                PERFIL PADRÃO AO ENTRAR
              </p>

              <div className="mt-3 space-y-2">
                {perfisEditando.map(
                  (
                    perfil
                  ) => (
                    <label
                      key={
                        perfil
                      }
                      className="flex cursor-pointer items-center gap-3 rounded-xl border border-blue-900 bg-blue-950/20 p-3"
                    >
                      <input
                        type="radio"
                        name="perfil-principal"
                        checked={
                          perfilPrincipalEditando ===
                          perfil
                        }
                        onChange={() =>
                          setPerfilPrincipalEditando(
                            perfil
                          )
                        }
                        className="h-5 w-5"
                      />

                      <span className="font-bold">
                        {nomePerfil(
                          perfil
                        )}
                      </span>
                    </label>
                  )
                )}
              </div>
            </section>

            <section className="mt-6">
              <p className="text-xs font-black text-slate-400">
                STATUS DO VÍNCULO
              </p>

              <div className="mt-3 grid grid-cols-2 gap-3">
                <button
                  type="button"
                  onClick={() =>
                    setAtivoEditando(
                      true
                    )
                  }
                  className={`rounded-xl border p-3 font-black ${
                    ativoEditando
                      ? "border-green-500 bg-green-950 text-green-300"
                      : "border-slate-700 bg-slate-800"
                  }`}
                >
                  🟢 Ativo
                </button>

                <button
                  type="button"
                  onClick={() =>
                    setAtivoEditando(
                      false
                    )
                  }
                  className={`rounded-xl border p-3 font-black ${
                    !ativoEditando
                      ? "border-red-500 bg-red-950 text-red-300"
                      : "border-slate-700 bg-slate-800"
                  }`}
                >
                  🔴 Inativo
                </button>
              </div>
            </section>

            <section className="mt-6">
              <p className="text-xs font-black text-slate-400">
                UNIDADES
              </p>

              {unidadesDoLocal.length ===
              0 ? (
                <div className="mt-3 rounded-xl border border-slate-700 bg-slate-800 p-4 text-sm text-slate-400">
                  Este local não possui unidades cadastradas.
                </div>
              ) : (
                <div className="mt-3 max-h-64 space-y-2 overflow-y-auto rounded-xl border border-slate-700 bg-slate-950 p-3">
                  {unidadesDoLocal.map(
                    (
                      unidade
                    ) => (
                      <label
                        key={
                          unidade.id
                        }
                        className="flex cursor-pointer items-center gap-3 rounded-xl border border-slate-800 bg-slate-900 p-3"
                      >
                        <input
                          type="checkbox"
                          checked={
                            unidadesEditando.includes(
                              unidade.id
                            )
                          }
                          onChange={() =>
                            alternarUnidade(
                              unidade.id
                            )
                          }
                          className="h-5 w-5"
                        />

                        <span className="font-bold">
                          {nomeUnidade(
                            unidade
                          )}
                        </span>
                      </label>
                    )
                  )}
                </div>
              )}
            </section>

            <div className="mt-6 grid grid-cols-2 gap-3">
              <button
                type="button"
                onClick={() =>
                  !salvando &&
                  setVinculoEditando(
                    null
                  )
                }
                className="rounded-xl bg-slate-700 py-3 font-black"
              >
                Cancelar
              </button>

              <button
                type="button"
                onClick={
                  salvarVinculo
                }
                disabled={
                  salvando
                }
                className="rounded-xl bg-green-600 py-3 font-black text-white disabled:opacity-50"
              >
                {salvando
                  ? "Salvando..."
                  : "💾 Salvar vínculo"}
              </button>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}
